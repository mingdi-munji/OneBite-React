### 기본적 세팅하기 (react 업데이트로 인한 추가적 세팅)
    //eslint.config.js를 추가하고 아래의 내용을 추가하여 작업을 편하게 함
        "no-unused-vars": "off", //안쓰는 변수에 대한 오류 표시에 대한 설정
        "react/prop-types": "off", //react 안정성에 대한 설정
    //맨 아래에 배열 추가 (rules 뒤) : Oxlint가 처리할 수 있는 규칙은 ESLint가 검사하지 않고 넘어가게 하여 성능 향상
        oxlint.configs['flat/recommended'], // 2. 맨 마지막 배열에 추가
    
    //두 도구가 엮일 수 있게 package.json를 수정
    "scripts": {
        "lint": "oxlint && eslint"
    }
